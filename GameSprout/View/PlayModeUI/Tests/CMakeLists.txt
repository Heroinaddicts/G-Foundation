cmake_minimum_required(VERSION 3.21)
project(PlayModeUIChecks LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../../Foundation/ParallelBuild.cmake")
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../IDEWindow" IDEWindow)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." PlayModeUI)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../GameViewUI" GameViewUI)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../SceneObjectUI" SceneObjectUI)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../ScriptUI" ScriptUI)
set_target_properties(IDEWindow PlayModeUI GameViewUI SceneObjectUI ScriptUI PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
add_executable(PlayModeUIChecks PlayModeUIChecks.cpp)
target_compile_features(PlayModeUIChecks PRIVATE cxx_std_20)
target_include_directories(PlayModeUIChecks PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}/.." "${CMAKE_CURRENT_LIST_DIR}/../../IDEWindow"
    "${CMAKE_CURRENT_LIST_DIR}/../../Interface" "${CMAKE_CURRENT_LIST_DIR}/../../../Module/Interface"
    "${CMAKE_CURRENT_LIST_DIR}/../../../GameSproutEditor/Api")
target_link_libraries(PlayModeUIChecks PRIVATE IDEWindow PlayModeUI Foundation::FoundationApi GameSproutRuntime::GameSproutRuntimeApi Qt6::Widgets Qt6::Test)
enable_testing()
add_test(NAME PlayModeUI.Toolbar COMMAND PlayModeUIChecks)
set_tests_properties(PlayModeUI.Toolbar PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)

add_executable(GameViewportInputChecks GameViewportInputChecks.cpp)
set_target_properties(GameViewportInputChecks PROPERTIES AUTOUIC ON AUTOUIC_SEARCH_PATHS "${CMAKE_CURRENT_LIST_DIR}/../../GameViewUI")
target_compile_features(GameViewportInputChecks PRIVATE cxx_std_20)
target_include_directories(GameViewportInputChecks PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../GameViewUI")
target_link_libraries(GameViewportInputChecks PRIVATE GameViewUI Foundation::FoundationApi GameSproutRuntime::GameSproutRuntimeApi Qt6::Widgets Qt6::Test)
add_test(NAME PlayModeUI.GameInput COMMAND GameViewportInputChecks)
set_tests_properties(PlayModeUI.GameInput PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)

set(GAMESPROUT_TEST_RUNTIME_DIRECTORY "" CACHE PATH "Optional independently built Runtime directory for native input integration")
if(GAMESPROUT_TEST_RUNTIME_DIRECTORY)
    add_test(NAME PlayModeUI.GameInputRuntime COMMAND GameViewportInputChecks "${GAMESPROUT_TEST_RUNTIME_DIRECTORY}")
    set_tests_properties(PlayModeUI.GameInputRuntime PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()

add_executable(InspectorValueChecks InspectorValueChecks.cpp)
target_compile_features(InspectorValueChecks PRIVATE cxx_std_20)
target_include_directories(InspectorValueChecks PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../SceneObjectUI")
target_link_libraries(InspectorValueChecks PRIVATE SceneObjectUI GameSproutInspectorControls Qt6::Widgets Qt6::Test)
add_test(NAME PlayModeUI.NarrowNumbers COMMAND InspectorValueChecks)
set_tests_properties(PlayModeUI.NarrowNumbers PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
if(GAMESPROUT_TEST_RUNTIME_DIRECTORY)
    add_test(NAME PlayModeUI.ReadOnlyInspector COMMAND InspectorValueChecks "${GAMESPROUT_TEST_RUNTIME_DIRECTORY}")
    set_tests_properties(PlayModeUI.ReadOnlyInspector PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()

add_executable(ScriptBuildStateChecks ScriptBuildStateChecks.cpp)
target_compile_features(ScriptBuildStateChecks PRIVATE cxx_std_20)
target_include_directories(ScriptBuildStateChecks PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../ScriptUI")
target_link_libraries(ScriptBuildStateChecks PRIVATE ScriptUI GameSproutInspectorControls Qt6::Widgets)
add_test(NAME PlayModeUI.ScriptBuildState COMMAND ScriptBuildStateChecks)
set_tests_properties(PlayModeUI.ScriptBuildState PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
