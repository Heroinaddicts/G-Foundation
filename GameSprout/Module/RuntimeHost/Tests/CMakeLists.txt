cmake_minimum_required(VERSION 3.21)
project(RuntimeHostChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${gamesprout_root}/Foundation/FoundationApi.cmake")
include("${gamesprout_root}/GameSproutRuntime/GameSproutRuntime.cmake")
find_package(Qt6 REQUIRED COMPONENTS Core)
set(GAMESPROUT_TEST_RUNTIME_DIRECTORY "${gamesprout_root}/../Bin" CACHE PATH "Runtime library to test")
add_executable(RuntimeHostChecks RuntimeHostChecks.cpp ../RuntimeHost.cpp)
target_compile_features(RuntimeHostChecks PRIVATE cxx_std_20)
target_include_directories(RuntimeHostChecks PRIVATE .. "${gamesprout_root}/Module/Interface" "${gamesprout_root}/GameSproutEditor/Api")
target_link_libraries(RuntimeHostChecks PRIVATE Foundation::FoundationApi GameSproutRuntime::GameSproutRuntimeApi Qt6::Core)
enable_testing()
add_test(NAME RuntimeHost.PlayPauseStep COMMAND RuntimeHostChecks "${GAMESPROUT_TEST_RUNTIME_DIRECTORY}")
set_tests_properties(RuntimeHost.PlayPauseStep PROPERTIES TIMEOUT 30)
