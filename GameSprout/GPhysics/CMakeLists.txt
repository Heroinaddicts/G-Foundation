cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(GPhysics VERSION 0.1 LANGUAGES CXX)
endif()
include(FetchContent)
# Jolt remains a private dependency. Pin both the release commit and archive
# checksum; no upstream source is placed in any Runtime/Editor source tree.
set(JPH_BUILD_SHARED_LIBS OFF CACHE BOOL "Private static Jolt" FORCE)
set(OVERRIDE_CXX_FLAGS OFF CACHE BOOL "Preserve host compiler options" FORCE)
set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "No cross-project LTO" FORCE)
set(ENABLE_ALL_WARNINGS OFF CACHE BOOL "Do not promote upstream warnings" FORCE)
set(DEBUG_RENDERER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "Runtime owns debug rendering" FORCE)
set(PROFILER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "Disable standalone Jolt profiler" FORCE)
set(GENERATE_DEBUG_SYMBOLS ON CACHE BOOL "Keep physics symbols" FORCE)
set(USE_STATIC_MSVC_RUNTIME_LIBRARY OFF CACHE BOOL "Match Runtime DLL's default MSVC CRT" FORCE)
foreach(compute JPH_USE_DX12 JPH_USE_VK JPH_USE_MTL JPH_USE_CPU_COMPUTE)
    set(${compute} OFF CACHE BOOL "Rigid body simulation uses CPU jobs" FORCE)
endforeach()
foreach(feature USE_SSE4_1 USE_SSE4_2 USE_AVX USE_AVX2 USE_AVX512 USE_LZCNT USE_TZCNT USE_F16C USE_FMADD)
    set(${feature} OFF CACHE BOOL "Portable CPU baseline" FORCE)
endforeach()
FetchContent_Declare(gamesprout_jolt
    URL "https://codeload.github.com/jrouwe/JoltPhysics/tar.gz/e77f175595e64cb44218cc9d9d56fc365ad0e36a"
    URL_HASH SHA256=1f32328fb763135de10a244568d6ccb2ed9b1e6593fafe6dc6db5b2719d330bd
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    SOURCE_SUBDIR Build)
FetchContent_MakeAvailable(gamesprout_jolt)
set_target_properties(Jolt PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC AND CMAKE_MSVC_RUNTIME_LIBRARY)
    set_target_properties(Jolt PROPERTIES MSVC_RUNTIME_LIBRARY "${CMAKE_MSVC_RUNTIME_LIBRARY}")
endif()
add_library(GPhysics STATIC GPhysics.h GPhysics.cpp MeshCooking.h MeshCooking.cpp MeshShapeCache.h MeshShapeCache.cpp)
add_library(GPhysics::GPhysics ALIAS GPhysics)
target_compile_features(GPhysics PUBLIC cxx_std_20)
target_include_directories(GPhysics PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(GPhysics PRIVATE Jolt)
set_target_properties(GPhysics PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
option(GPHYSICS_BUILD_TESTS "Build standalone Jolt backend checks" OFF)
if(GPHYSICS_BUILD_TESTS)
    enable_testing()
    add_subdirectory(Tests)
endif()
