cmake_minimum_required(VERSION 3.21)
project(SceneEditingChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
get_filename_component(repository_root "${gamesprout_root}/.." ABSOLUTE)
include("${gamesprout_root}/Foundation/FoundationApi.cmake")
include("${gamesprout_root}/GRHI/GRHI.cmake")
include("${gamesprout_root}/GameSproutRuntime/GameSproutRuntime.cmake")
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
if(APPLE)
    set(scene_test_binary_root "${repository_root}/Bin/GameSproutEditor.app/Contents/MacOS")
else()
    set(scene_test_binary_root "${repository_root}/Bin")
endif()
add_executable(SceneTransformEditChecks SceneTransformEditChecks.cpp SceneEditingTestSupport.h)
add_executable(SceneInteractionChecks SceneInteractionChecks.cpp SceneInteractionFixture.h
    "${gamesprout_root}/View/SceneViewUI/SceneGizmo.cpp" "${gamesprout_root}/View/SceneViewUI/SceneGizmoGeometry.cpp")
add_executable(SceneDeletionChecks SceneDeletionChecks.cpp SceneInteractionFixture.h
    "${gamesprout_root}/View/SceneViewUI/SceneGizmo.cpp" "${gamesprout_root}/View/SceneViewUI/SceneGizmoGeometry.cpp")
add_library(SceneViewUIEditingUnderTest SHARED IMPORTED)
if(WIN32)
    set_target_properties(SceneViewUIEditingUnderTest PROPERTIES IMPORTED_LOCATION "${scene_test_binary_root}/Logic/View/SceneViewUI.dll" IMPORTED_IMPLIB "${scene_test_binary_root}/Logic/View/SceneViewUI.lib")
else()
    set_target_properties(SceneViewUIEditingUnderTest PROPERTIES IMPORTED_LOCATION "${scene_test_binary_root}/Logic/View/${CMAKE_SHARED_LIBRARY_PREFIX}SceneViewUI${CMAKE_SHARED_LIBRARY_SUFFIX}")
endif()
target_link_libraries(SceneInteractionChecks PRIVATE SceneViewUIEditingUnderTest)
target_link_libraries(SceneDeletionChecks PRIVATE SceneViewUIEditingUnderTest)
foreach(target IN ITEMS SceneTransformEditChecks SceneInteractionChecks SceneDeletionChecks)
target_compile_features(${target} PRIVATE cxx_std_20)
target_include_directories(${target} PRIVATE
    "${gamesprout_root}/Module/Interface" "${gamesprout_root}/View/Interface" "${gamesprout_root}/GameSproutEditor/Api" "${gamesprout_root}/View/SceneViewUI")
target_link_libraries(${target} PRIVATE Foundation::FoundationApi GameSproutRuntime::GameSproutRuntimeApi Qt6::Widgets Qt6::Test)
target_compile_definitions(${target} PRIVATE TEST_BINARY_ROOT="${scene_test_binary_root}" TEST_RUNTIME_ROOT="${repository_root}/Bin")
set_property(TARGET ${target} APPEND PROPERTY LINK_DEPENDS "$<TARGET_FILE:GameSproutRuntime::GameSproutRuntime>")
add_custom_command(TARGET ${target} POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:GameSproutRuntime::GameSproutRuntime>"
        "$<TARGET_FILE_DIR:${target}>/$<TARGET_FILE_NAME:GameSproutRuntime::GameSproutRuntime>"
    VERBATIM)
endforeach()
enable_testing()
add_test(NAME SceneEditing.TransformTransaction COMMAND SceneTransformEditChecks)
set_tests_properties(SceneEditing.TransformTransaction PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
add_test(NAME SceneEditing.NativeInteraction COMMAND SceneInteractionChecks)
set_tests_properties(SceneEditing.NativeInteraction PROPERTIES TIMEOUT 90)
add_test(NAME SceneEditing.DeleteSubtree COMMAND SceneDeletionChecks --module-only)
set_tests_properties(SceneEditing.DeleteSubtree PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
add_test(NAME SceneEditing.NativeDeletion COMMAND SceneDeletionChecks)
set_tests_properties(SceneEditing.NativeDeletion PROPERTIES TIMEOUT 90)
