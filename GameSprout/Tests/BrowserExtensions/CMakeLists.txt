cmake_minimum_required(VERSION 3.21)
project(BrowserExtensionChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${gamesprout_root}/Foundation/FoundationApi.cmake")
include("${gamesprout_root}/GameSproutRuntime/GameSproutRuntime.cmake")
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_subdirectory("${gamesprout_root}/View/ProjectAssetsUI" ProjectAssetsUI)
add_subdirectory("${gamesprout_root}/View/HierarchyUI" HierarchyUI)
set_target_properties(ProjectAssetsUI HierarchyUI PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/views")
add_executable(BrowserExtensionChecks BrowserExtensionChecks.cpp)
target_compile_features(BrowserExtensionChecks PRIVATE cxx_std_20)
target_include_directories(BrowserExtensionChecks PRIVATE "${gamesprout_root}/View/Interface"
    "${gamesprout_root}/Module/Interface" "${gamesprout_root}/GameSproutEditor/Api")
target_link_libraries(BrowserExtensionChecks PRIVATE Foundation::FoundationApi GameSproutRuntime::GameSproutRuntimeApi Qt6::Widgets Qt6::Test ${CMAKE_DL_LIBS})
target_compile_definitions(BrowserExtensionChecks PRIVATE PROJECT_VIEW_FILE="$<TARGET_FILE:ProjectAssetsUI>" HIERARCHY_VIEW_FILE="$<TARGET_FILE:HierarchyUI>")
add_dependencies(BrowserExtensionChecks ProjectAssetsUI HierarchyUI)
set(BROWSER_TEST_RUNTIME_DIRECTORY "${gamesprout_root}/../Bin" CACHE PATH "Independent Runtime SDK directory")
enable_testing()
add_test(NAME BrowserExtensions.ValueDropsAndContext COMMAND BrowserExtensionChecks "${BROWSER_TEST_RUNTIME_DIRECTORY}")
set_tests_properties(BrowserExtensions.ValueDropsAndContext PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
