cmake_minimum_required(VERSION 3.21)
project(ProjectSettingsUIChecks LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../Foundation/ParallelBuild.cmake")
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${gamesprout_root}/View/ProjectSettingsUI" ProjectSettingsUI)
add_subdirectory("${gamesprout_root}/View/SceneObjectUI" SceneObjectUI)
add_subdirectory("${gamesprout_root}/Module/SceneManager" SceneManager)
set_target_properties(ProjectSettingsUI SceneObjectUI SceneManager PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/out")
add_executable(ProjectSettingsUIChecks ProjectSettingsUIChecks.cpp)
target_compile_features(ProjectSettingsUIChecks PRIVATE cxx_std_20)
target_include_directories(ProjectSettingsUIChecks PRIVATE "${gamesprout_root}/View/ProjectSettingsUI" "${gamesprout_root}/View/SceneObjectUI" "${gamesprout_root}/Module/SceneManager" "${gamesprout_root}/ObjectDocument")
target_link_libraries(ProjectSettingsUIChecks PRIVATE ProjectSettingsUI SceneObjectUI SceneManager GameSproutInspectorControls Qt6::Widgets Qt6::Test ${CMAKE_DL_LIBS})
set(PROJECT_SETTINGS_RUNTIME_DIRECTORY "${gamesprout_root}/../Bin" CACHE PATH "Runtime directory for UI checks")
enable_testing()
add_test(NAME ProjectSettingsUI.DefinitionsAndInspector COMMAND ProjectSettingsUIChecks "${PROJECT_SETTINGS_RUNTIME_DIRECTORY}")
set_tests_properties(ProjectSettingsUI.DefinitionsAndInspector PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
