cmake_minimum_required(VERSION 3.21)
project(AnimationAssetsChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
get_filename_component(repository_root "${gamesprout_root}/.." ABSOLUTE)
include("${gamesprout_root}/Foundation/FoundationApi.cmake")
include("${gamesprout_root}/GRHI/GRHI.cmake")
include("${gamesprout_root}/GameSproutRuntime/GameSproutRuntime.cmake")
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
if(APPLE)
    set(module_root "${repository_root}/Bin/GameSproutEditor.app/Contents/MacOS/Logic/Module")
else()
    set(module_root "${repository_root}/Bin/Logic/Module")
endif()
foreach(name IN ITEMS AssetsPipeline RuntimeAssets)
    add_library(${name} SHARED IMPORTED)
    set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${module_root}/${CMAKE_SHARED_LIBRARY_PREFIX}${name}${CMAKE_SHARED_LIBRARY_SUFFIX}")
    if(WIN32)
        set_target_properties(${name} PROPERTIES IMPORTED_IMPLIB "${module_root}/${name}.lib")
    endif()
endforeach()
add_executable(AnimationAssetsChecks AnimationAssetsChecks.cpp)
target_compile_features(AnimationAssetsChecks PRIVATE cxx_std_20)
target_compile_definitions(AnimationAssetsChecks PRIVATE TEST_BINARY_ROOT="${module_root}/../..")
target_include_directories(AnimationAssetsChecks PRIVATE "${gamesprout_root}/Module/AssetsPipeline" "${gamesprout_root}/Module/Interface"
    "${gamesprout_root}/GameSproutEditor/Api" "${gamesprout_root}/View/Interface")
target_link_libraries(AnimationAssetsChecks PRIVATE AssetsPipeline RuntimeAssets Foundation::FoundationApi GRHI::GRHIApi
    GameSproutRuntime::GameSproutRuntimeApi Qt6::Core Qt6::Gui)
set_target_properties(AnimationAssetsChecks PROPERTIES BUILD_RPATH "${module_root}")
enable_testing()
add_test(NAME Assets.AnimationFormat COMMAND AnimationAssetsChecks)
set_tests_properties(Assets.AnimationFormat PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_library(InspectorUI SHARED IMPORTED)
get_filename_component(view_root "${module_root}/../View" ABSOLUTE)
set_target_properties(InspectorUI PROPERTIES IMPORTED_LOCATION "${view_root}/${CMAKE_SHARED_LIBRARY_PREFIX}InspectorUI${CMAKE_SHARED_LIBRARY_SUFFIX}")
if(WIN32)
    set_target_properties(InspectorUI PROPERTIES IMPORTED_IMPLIB "${view_root}/InspectorUI.lib")
endif()
add_executable(ControllerDialogChecks ControllerDialogChecks.cpp)
target_compile_features(ControllerDialogChecks PRIVATE cxx_std_20)
target_include_directories(ControllerDialogChecks PRIVATE "${gamesprout_root}/Module/Interface" "${gamesprout_root}/View/Interface" "${gamesprout_root}/GameSproutEditor/Api")
target_link_libraries(ControllerDialogChecks PRIVATE InspectorUI Foundation::FoundationApi GRHI::GRHIApi GameSproutRuntime::GameSproutRuntimeApi Qt6::Widgets)
set_target_properties(ControllerDialogChecks PROPERTIES BUILD_RPATH "${view_root}")
add_test(NAME IDE.AnimatorControllerDialog COMMAND ControllerDialogChecks)
set_tests_properties(IDE.AnimatorControllerDialog PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
