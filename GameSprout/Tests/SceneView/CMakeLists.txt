cmake_minimum_required(VERSION 3.21)
project(SceneViewChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
get_filename_component(repository_root "${gamesprout_root}/.." ABSOLUTE)
include("${gamesprout_root}/Foundation/FoundationApi.cmake")
include("${gamesprout_root}/GRHI/GRHI.cmake")
include("${gamesprout_root}/GameSproutRuntime/GameSproutRuntime.cmake")
find_package(Qt6 REQUIRED COMPONENTS Widgets)
if(APPLE)
    set(scene_test_binary_root "${repository_root}/Bin/GameSproutEditor.app/Contents/MacOS")
else()
    set(scene_test_binary_root "${repository_root}/Bin")
endif()
add_executable(SceneCameraChecks SceneCameraChecks.cpp "${gamesprout_root}/View/SceneViewUI/SceneViewCamera.cpp")
add_executable(SceneViewportChecks SceneViewportChecks.cpp)
add_executable(SceneGizmoChecks SceneGizmoChecks.cpp
    "${gamesprout_root}/View/SceneViewUI/SceneGizmo.cpp"
    "${gamesprout_root}/View/SceneViewUI/SceneGizmoGeometry.cpp")
target_compile_features(SceneGizmoChecks PRIVATE cxx_std_20)
target_include_directories(SceneGizmoChecks PRIVATE "${gamesprout_root}/View/SceneViewUI")
target_link_libraries(SceneGizmoChecks PRIVATE GameSproutRuntime::GameSproutRuntimeApi)
add_library(SceneViewUIUnderTest SHARED IMPORTED)
if(WIN32)
    set_target_properties(SceneViewUIUnderTest PROPERTIES IMPORTED_LOCATION "${scene_test_binary_root}/Logic/View/SceneViewUI.dll" IMPORTED_IMPLIB "${scene_test_binary_root}/Logic/View/SceneViewUI.lib")
else()
    set_target_properties(SceneViewUIUnderTest PROPERTIES IMPORTED_LOCATION "${scene_test_binary_root}/Logic/View/${CMAKE_SHARED_LIBRARY_PREFIX}SceneViewUI${CMAKE_SHARED_LIBRARY_SUFFIX}")
endif()
target_link_libraries(SceneViewportChecks PRIVATE SceneViewUIUnderTest)
foreach(target IN ITEMS SceneCameraChecks SceneViewportChecks)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${gamesprout_root}/View/SceneViewUI" "${gamesprout_root}/View/Interface" "${gamesprout_root}/Module/Interface" "${gamesprout_root}/GameSproutEditor/Api")
    target_link_libraries(${target} PRIVATE Foundation::FoundationApi GameSproutRuntime::GameSproutRuntimeApi Qt6::Widgets)
    target_compile_definitions(${target} PRIVATE TEST_BINARY_ROOT="${scene_test_binary_root}" TEST_RUNTIME_ROOT="${repository_root}/Bin")
endforeach()
enable_testing()
add_test(NAME SceneView.Camera COMMAND SceneCameraChecks)
add_test(NAME SceneView.Gizmo COMMAND SceneGizmoChecks)
add_test(NAME SceneView.Headless COMMAND SceneViewportChecks --headless)
set_tests_properties(SceneView.Headless PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
add_test(NAME SceneView.Native COMMAND SceneViewportChecks)
set_tests_properties(SceneView.Native PROPERTIES TIMEOUT 60)
