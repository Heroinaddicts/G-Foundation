cmake_minimum_required(VERSION 3.21)
project(PrimitiveChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${gamesprout_root}/Foundation/ParallelBuild.cmake")
find_package(Qt6 REQUIRED COMPONENTS Widgets)
foreach(module IN ITEMS SceneManager PrimitiveAuthoring)
 add_subdirectory("${gamesprout_root}/Module/${module}" "${module}")
 set_target_properties(${module} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
endforeach()
add_subdirectory("${gamesprout_root}/View/PrimitiveUI" PrimitiveUI)
set_target_properties(PrimitiveUI PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
include("${gamesprout_root}/View/Shared/InspectorControls/InspectorControls.cmake")
add_executable(PrimitiveGeometryChecks PrimitiveGeometryChecks.cpp "${gamesprout_root}/GameSproutRuntime/Resources/BuiltinMeshes.cpp")
target_compile_features(PrimitiveGeometryChecks PRIVATE cxx_std_20)
target_include_directories(PrimitiveGeometryChecks PRIVATE "${gamesprout_root}/GameSproutRuntime")
target_link_libraries(PrimitiveGeometryChecks PRIVATE GameSproutRuntime::GameSproutRuntimeApi)
add_executable(PrimitiveAuthoringChecks PrimitiveAuthoringChecks.cpp)
target_compile_features(PrimitiveAuthoringChecks PRIVATE cxx_std_20)
target_include_directories(PrimitiveAuthoringChecks PRIVATE "${gamesprout_root}/Module/PrimitiveAuthoring" "${gamesprout_root}/Module/SceneManager" "${gamesprout_root}/View/PrimitiveUI" "${gamesprout_root}/ObjectDocument")
target_link_libraries(PrimitiveAuthoringChecks PRIVATE SceneManager PrimitiveAuthoring PrimitiveUI ObjectDocument::ObjectDocument GameSproutInspectorControls Qt6::Widgets)
add_executable(PrimitiveRenderChecks PrimitiveRenderChecks.cpp)
target_compile_features(PrimitiveRenderChecks PRIVATE cxx_std_20)
target_link_libraries(PrimitiveRenderChecks PRIVATE GameSproutRuntime::GameSproutRuntimeApi)
set(PRIMITIVE_RUNTIME_DIRECTORY "${gamesprout_root}/../Bin" CACHE PATH "Directory containing a current independent Runtime")
enable_testing()
add_test(NAME Primitives.Geometry COMMAND PrimitiveGeometryChecks)
add_test(NAME Primitives.AuthoringAndPersistence COMMAND PrimitiveAuthoringChecks "${PRIMITIVE_RUNTIME_DIRECTORY}")
set_tests_properties(Primitives.AuthoringAndPersistence PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
add_test(NAME Primitives.RenderGPU COMMAND PrimitiveRenderChecks "${PRIMITIVE_RUNTIME_DIRECTORY}")
set_tests_properties(Primitives.RenderGPU PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 45)
