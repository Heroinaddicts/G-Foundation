cmake_minimum_required(VERSION 3.21)
project(StaticAuthoringChecks LANGUAGES CXX)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${gamesprout_root}/Module/StaticAuthoring" StaticAuthoring)
set_target_properties(StaticAuthoring PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
add_executable(StaticAuthoringChecks StaticAuthoringChecks.cpp)
target_compile_features(StaticAuthoringChecks PRIVATE cxx_std_20)
target_include_directories(StaticAuthoringChecks PRIVATE "${gamesprout_root}/Module/StaticAuthoring" "${gamesprout_root}/Module/Interface" "${gamesprout_root}/GameSproutEditor/Api")
target_link_libraries(StaticAuthoringChecks PRIVATE StaticAuthoring Foundation::FoundationApi GameSproutRuntime::GameSproutRuntimeApi Qt${QT_VERSION_MAJOR}::Core)
target_compile_definitions(StaticAuthoringChecks PRIVATE RUNTIME_LIBRARY_DIRECTORY="$<TARGET_FILE_DIR:GameSproutRuntime::GameSproutRuntime>")
enable_testing()
add_test(NAME StaticAuthoring.CapabilitiesAndPolicy COMMAND StaticAuthoringChecks)
