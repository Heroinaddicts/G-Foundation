cmake_minimum_required(VERSION 3.21)
project(ObjectEditingChecks LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${gamesprout_root}/Module/ObjectEditing" ObjectEditing)
include("${gamesprout_root}/ObjectDocument/ObjectDocument.cmake")
set_target_properties(ObjectEditing PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modules" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/modules")
add_executable(ObjectEditingChecks ObjectEditingChecks.cpp)
target_compile_features(ObjectEditingChecks PRIVATE cxx_std_20)
target_include_directories(ObjectEditingChecks PRIVATE "${gamesprout_root}/Module/ObjectEditing" "${gamesprout_root}/Module/Interface" "${gamesprout_root}/GameSproutEditor/Api")
target_link_libraries(ObjectEditingChecks PRIVATE ObjectEditing Foundation::FoundationApi GameSproutRuntime::GameSproutRuntimeApi Qt6::Core)
set(OBJECT_EDITING_RUNTIME_DIRECTORY "${gamesprout_root}/../Bin" CACHE PATH "Independently built Runtime matching this SDK")
enable_testing()
add_test(NAME ObjectEditing.ModeIdentityAndRollback COMMAND ObjectEditingChecks "${OBJECT_EDITING_RUNTIME_DIRECTORY}")
add_executable(TransientReferenceDocumentChecks TransientReferenceDocumentChecks.cpp)
target_compile_features(TransientReferenceDocumentChecks PRIVATE cxx_std_20)
target_link_libraries(TransientReferenceDocumentChecks PRIVATE ObjectDocument::ObjectDocument GameSproutRuntime::GameSproutRuntimeApi)
add_test(NAME ObjectEditing.TransientReferencesNeverPersist COMMAND TransientReferenceDocumentChecks)
