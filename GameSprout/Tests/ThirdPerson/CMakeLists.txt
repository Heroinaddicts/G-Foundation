cmake_minimum_required(VERSION 3.21)
project(ThirdPersonChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${gamesprout_root}/Foundation/FoundationApi.cmake")
include("${gamesprout_root}/GameSproutRuntime/GameSproutRuntime.cmake")
include("${gamesprout_root}/ObjectDocument/ObjectDocument.cmake")
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(module_root "${gamesprout_root}/../Bin/GameSproutEditor.app/Contents/MacOS/Logic/Module" CACHE PATH "Independently built modules")
set(THIRDPERSON_TEST_RUNTIME_DIRECTORY "${gamesprout_root}/../Bin" CACHE PATH "Independently built Runtime SDK directory")
foreach(name IN ITEMS ScriptCompiler AssetsPipeline RuntimeAssets)
    add_library(${name} SHARED IMPORTED)
    set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${module_root}/${CMAKE_SHARED_LIBRARY_PREFIX}${name}${CMAKE_SHARED_LIBRARY_SUFFIX}")
endforeach()
add_executable(ThirdPersonChecks ThirdPersonChecks.cpp)
target_compile_features(ThirdPersonChecks PRIVATE cxx_std_20)
target_include_directories(ThirdPersonChecks PRIVATE "${gamesprout_root}/Module/RuntimeAssets" "${gamesprout_root}/Module/ScriptCompiler" "${gamesprout_root}/Module/AssetsPipeline" "${gamesprout_root}/Module/Interface" "${gamesprout_root}/GameSproutEditor/Api" "${gamesprout_root}/View/Interface")
target_link_libraries(ThirdPersonChecks PRIVATE ScriptCompiler AssetsPipeline RuntimeAssets ObjectDocument::ObjectDocument Foundation::FoundationApi GameSproutRuntime::GameSproutRuntimeApi Qt6::Core Qt6::Gui)
target_compile_definitions(ThirdPersonChecks PRIVATE TEST_GENERATOR="${gamesprout_root}/Examples/ThirdPerson/CreateProject.py" TEST_PYTHON="${Python3_EXECUTABLE}" TEST_RUNTIME="${THIRDPERSON_TEST_RUNTIME_DIRECTORY}")
set_target_properties(ThirdPersonChecks PROPERTIES BUILD_RPATH "${module_root}")
enable_testing()
add_test(NAME Gameplay.ThirdPerson COMMAND ThirdPersonChecks)
set_tests_properties(Gameplay.ThirdPerson PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
