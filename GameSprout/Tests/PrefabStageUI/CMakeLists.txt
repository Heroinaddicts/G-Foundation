cmake_minimum_required(VERSION 3.21)
project(PrefabStageUIChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${gamesprout_root}/Foundation/FoundationApi.cmake")
include("${gamesprout_root}/GameSproutRuntime/GameSproutRuntime.cmake")
find_package(Qt6 REQUIRED COMPONENTS Widgets)
file(GLOB stage_ui_files CONFIGURE_DEPENDS "${gamesprout_root}/View/PrefabStageUI/*.cpp" "${gamesprout_root}/View/PrefabStageUI/*.h" "${gamesprout_root}/View/PrefabStageUI/*.ui")
list(FILTER stage_ui_files EXCLUDE REGEX "/ExportView[.]cpp$")
file(GLOB command_ui_files CONFIGURE_DEPENDS "${gamesprout_root}/View/SceneEditorUI/*.cpp" "${gamesprout_root}/View/SceneEditorUI/*.h" "${gamesprout_root}/View/SceneEditorUI/*.ui")
list(FILTER command_ui_files EXCLUDE REGEX "/ExportView[.]cpp$")
file(GLOB ide_ui_files CONFIGURE_DEPENDS "${gamesprout_root}/View/IDEWindow/*.cpp" "${gamesprout_root}/View/IDEWindow/*.h" "${gamesprout_root}/View/IDEWindow/*.ui")
list(FILTER ide_ui_files EXCLUDE REGEX "/ExportView[.]cpp$")
add_executable(PrefabStageUIChecks PrefabStageUIChecks.cpp StageUIFixtures.h ${stage_ui_files} ${command_ui_files} ${ide_ui_files})
target_compile_features(PrefabStageUIChecks PRIVATE cxx_std_20)
target_include_directories(PrefabStageUIChecks PRIVATE "${gamesprout_root}/View/PrefabStageUI" "${gamesprout_root}/View/SceneEditorUI" "${gamesprout_root}/View/IDEWindow"
    "${gamesprout_root}/View/Interface" "${gamesprout_root}/Module/Interface" "${gamesprout_root}/GameSproutEditor/Api")
target_link_libraries(PrefabStageUIChecks PRIVATE Qt6::Widgets Foundation::FoundationApi GameSproutRuntime::GameSproutRuntimeApi)
set_target_properties(PrefabStageUIChecks PROPERTIES AUTOMOC ON AUTOUIC ON AUTOUIC_SEARCH_PATHS "${gamesprout_root}/View/PrefabStageUI;${gamesprout_root}/View/SceneEditorUI;${gamesprout_root}/View/IDEWindow")
enable_testing()
add_test(NAME PrefabStageUI.CommandsAndNavigation COMMAND PrefabStageUIChecks)
set_tests_properties(PrefabStageUI.CommandsAndNavigation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
