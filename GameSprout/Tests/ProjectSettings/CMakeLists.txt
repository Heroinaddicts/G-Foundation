cmake_minimum_required(VERSION 3.21)
project(ProjectSettingsChecks LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${gamesprout_root}/Module/ProjectSettings" ProjectSettings)
set_target_properties(ProjectSettings PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
set(GAMESPROUT_TEST_RUNTIME_LIBRARY "" CACHE FILEPATH "Independent Runtime library for verification")
if(GAMESPROUT_TEST_RUNTIME_LIBRARY)
    set_target_properties(GameSproutRuntime::GameSproutRuntime PROPERTIES IMPORTED_LOCATION "${GAMESPROUT_TEST_RUNTIME_LIBRARY}")
endif()
add_executable(ProjectSettingsChecks ProjectSettingsChecks.cpp)
target_compile_features(ProjectSettingsChecks PRIVATE cxx_std_20)
target_include_directories(ProjectSettingsChecks PRIVATE "${gamesprout_root}/Module/ProjectSettings" "${gamesprout_root}/Module/Interface" "${gamesprout_root}/GameSproutEditor/Api")
target_link_libraries(ProjectSettingsChecks PRIVATE ProjectSettings Foundation::FoundationApi GameSproutRuntime::GameSproutRuntimeApi Qt6::Core)
target_compile_definitions(ProjectSettingsChecks PRIVATE RUNTIME_LIBRARY_DIRECTORY="$<TARGET_FILE_DIR:GameSproutRuntime::GameSproutRuntime>")
enable_testing()
add_test(NAME ProjectSettings.CurrentFormatTransactions COMMAND ProjectSettingsChecks)
