cmake_minimum_required(VERSION 3.21)
project(AnimationMotionEditorChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
get_filename_component(repository_root "${gamesprout_root}/.." ABSOLUTE)
include("${gamesprout_root}/Foundation/FoundationApi.cmake")
include("${gamesprout_root}/GRHI/GRHI.cmake")
include("${gamesprout_root}/GameSproutRuntime/GameSproutRuntime.cmake")
find_package(Qt6 REQUIRED COMPONENTS Widgets)
if(APPLE)
    set(view_root "${repository_root}/Bin/GameSproutEditor.app/Contents/MacOS/Logic/View")
else()
    set(view_root "${repository_root}/Bin/Logic/View")
endif()
add_library(InspectorUI SHARED IMPORTED)
set_target_properties(InspectorUI PROPERTIES IMPORTED_LOCATION "${view_root}/${CMAKE_SHARED_LIBRARY_PREFIX}InspectorUI${CMAKE_SHARED_LIBRARY_SUFFIX}")
if(WIN32)
    set_target_properties(InspectorUI PROPERTIES IMPORTED_IMPLIB "${view_root}/InspectorUI.lib")
endif()
add_executable(AnimationMotionEditorChecks AnimationMotionEditorChecks.cpp)
target_compile_features(AnimationMotionEditorChecks PRIVATE cxx_std_20)
target_include_directories(AnimationMotionEditorChecks PRIVATE "${gamesprout_root}/Module/Interface" "${gamesprout_root}/View/Interface" "${gamesprout_root}/GameSproutEditor/Api")
target_link_libraries(AnimationMotionEditorChecks PRIVATE InspectorUI Foundation::FoundationApi GRHI::GRHIApi GameSproutRuntime::GameSproutRuntimeApi Qt6::Widgets)
set_target_properties(AnimationMotionEditorChecks PROPERTIES BUILD_RPATH "${view_root}")
enable_testing()
add_test(NAME AnimationMotionEditor.ImportSettingsDraft COMMAND AnimationMotionEditorChecks)
set_tests_properties(AnimationMotionEditor.ImportSettingsDraft PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
