cmake_minimum_required(VERSION 3.21)
project(PhysicsUIChecks LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../../Foundation/ParallelBuild.cmake")
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${gamesprout_root}/View/PhysicsUI" PhysicsUI)
add_subdirectory("${gamesprout_root}/View/SceneObjectUI" SceneObjectUI)
set_target_properties(PhysicsUI SceneObjectUI PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/out")
add_executable(PhysicsUIChecks PhysicsUIChecks.cpp)
target_compile_features(PhysicsUIChecks PRIVATE cxx_std_20)
target_compile_definitions(PhysicsUIChecks PRIVATE PHYSICS_INSPECTOR_UI_FILE="${gamesprout_root}/View/InspectorUI/InspectorUI.ui")
target_include_directories(PhysicsUIChecks PRIVATE "${gamesprout_root}/View/PhysicsUI" "${gamesprout_root}/View/SceneObjectUI")
target_link_libraries(PhysicsUIChecks PRIVATE PhysicsUI SceneObjectUI GameSproutInspectorControls Qt6::Widgets Qt6::Test ${CMAKE_DL_LIBS})
set(PHYSICS_TEST_RUNTIME_DIRECTORY "${gamesprout_root}/../Bin" CACHE PATH "Runtime directory for isolated UI checks")
enable_testing()
add_test(NAME PhysicsUI.ComponentsAndMaterial COMMAND PhysicsUIChecks "${PHYSICS_TEST_RUNTIME_DIRECTORY}")
set_tests_properties(PhysicsUI.ComponentsAndMaterial PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
