cmake_minimum_required(VERSION 3.21)
project(ScriptDocumentChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${gamesprout_root}/Foundation/ParallelBuild.cmake")
include("${gamesprout_root}/ObjectDocument/ObjectDocument.cmake")
add_executable(ScriptDocumentChecks ScriptDocumentChecks.cpp)
target_compile_features(ScriptDocumentChecks PRIVATE cxx_std_20)
target_include_directories(ScriptDocumentChecks PRIVATE "${gamesprout_root}/Module/Shared")
target_link_libraries(ScriptDocumentChecks PRIVATE ObjectDocument::ObjectDocument)
enable_testing()
set(SCRIPT_DOCUMENT_RUNTIME_DIRECTORY "${gamesprout_root}/../Bin" CACHE PATH "Current independent Runtime")
add_test(NAME ScriptDocuments.RecursiveReferences COMMAND ScriptDocumentChecks "${SCRIPT_DOCUMENT_RUNTIME_DIRECTORY}")
