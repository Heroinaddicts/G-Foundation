cmake_minimum_required(VERSION 3.21)
project(PhysicsDocumentChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${gamesprout_root}/ObjectDocument/ObjectDocument.cmake")
set(GAMESPROUT_TEST_RUNTIME_LIBRARY "" CACHE FILEPATH "Independent Runtime library for verification")
if(GAMESPROUT_TEST_RUNTIME_LIBRARY)
    set_target_properties(GameSproutRuntime::GameSproutRuntime PROPERTIES IMPORTED_LOCATION "${GAMESPROUT_TEST_RUNTIME_LIBRARY}")
endif()
add_executable(PhysicsMaterialChecks PhysicsMaterialChecks.cpp)
target_compile_features(PhysicsMaterialChecks PRIVATE cxx_std_20)
target_include_directories(PhysicsMaterialChecks PRIVATE "${gamesprout_root}/Module/Shared")
target_link_libraries(PhysicsMaterialChecks PRIVATE ObjectDocumentJson GameSproutRuntime::GameSproutRuntimeApi)
add_executable(PhysicsDocumentChecks PhysicsDocumentChecks.cpp)
target_compile_features(PhysicsDocumentChecks PRIVATE cxx_std_20)
target_link_libraries(PhysicsDocumentChecks PRIVATE ObjectDocument::ObjectDocument GameSproutRuntime::GameSproutRuntime)
target_compile_definitions(PhysicsDocumentChecks PRIVATE RUNTIME_LIBRARY_DIRECTORY="$<TARGET_FILE_DIR:GameSproutRuntime::GameSproutRuntime>")
enable_testing()
add_test(NAME Physics.MaterialCurrentFormat COMMAND PhysicsMaterialChecks)
add_test(NAME Physics.ComponentDocuments COMMAND PhysicsDocumentChecks)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_subdirectory("${gamesprout_root}/Module/PhysicsAuthoring" PhysicsAuthoring)
set_target_properties(PhysicsAuthoring PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
add_executable(PhysicsAuthoringChecks PhysicsAuthoringChecks.cpp)
target_compile_features(PhysicsAuthoringChecks PRIVATE cxx_std_20)
target_include_directories(PhysicsAuthoringChecks PRIVATE "${gamesprout_root}/Module/PhysicsAuthoring" "${gamesprout_root}/Module/Interface" "${gamesprout_root}/GameSproutEditor/Api")
target_link_libraries(PhysicsAuthoringChecks PRIVATE PhysicsAuthoring Foundation::FoundationApi Qt6::Core GameSproutRuntime::GameSproutRuntimeApi)
add_test(NAME Physics.MaterialSourceTransactions COMMAND PhysicsAuthoringChecks)
