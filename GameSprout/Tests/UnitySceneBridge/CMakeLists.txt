cmake_minimum_required(VERSION 3.21)
project(UnitySceneBridgeChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${gamesprout_root}/Foundation/ParallelBuild.cmake")
include("${gamesprout_root}/Foundation/FoundationApi.cmake")
include("${gamesprout_root}/GameSproutRuntime/GameSproutRuntime.cmake")
include("${gamesprout_root}/ObjectDocument/ObjectDocument.cmake")
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
if(APPLE)
    set(bridge_binary_root "${gamesprout_root}/../Bin/GameSproutEditor.app/Contents/MacOS")
else()
    set(bridge_binary_root "${gamesprout_root}/../Bin")
endif()
set(BRIDGE_TEST_BINARY_ROOT "${bridge_binary_root}" CACHE PATH "Built Editor plugins and Runtime")
add_executable(UnitySceneBridgeChecks UnitySceneBridgeChecks.cpp)
target_compile_features(UnitySceneBridgeChecks PRIVATE cxx_std_20)
target_compile_definitions(UnitySceneBridgeChecks PRIVATE TEST_BINARY_ROOT="${BRIDGE_TEST_BINARY_ROOT}")
target_include_directories(UnitySceneBridgeChecks PRIVATE
    "${gamesprout_root}/Module/Interface" "${gamesprout_root}/View/Interface"
    "${gamesprout_root}/GameSproutEditor/Api")
target_link_libraries(UnitySceneBridgeChecks PRIVATE ObjectDocument::ObjectDocument
    Foundation::FoundationApi GameSproutRuntime::GameSproutRuntimeApi Qt6::Core Qt6::Gui)
enable_testing()
add_test(NAME UnityBridge.NativeProjectPipeline COMMAND UnitySceneBridgeChecks --fixture)
set_tests_properties(UnityBridge.NativeProjectPipeline PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
