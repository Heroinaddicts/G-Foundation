cmake_minimum_required(VERSION 3.21)
project(PrefabPipelineChecks LANGUAGES CXX)
get_filename_component(gamesprout_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
get_filename_component(repository_root "${gamesprout_root}/.." ABSOLUTE)
include("${gamesprout_root}/Foundation/FoundationApi.cmake")
include("${gamesprout_root}/ObjectDocument/ObjectDocument.cmake")
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
if(APPLE)
    set(default_pipeline "${repository_root}/Bin/GameSproutEditor.app/Contents/MacOS/Logic/Module/libAssetsPipeline.dylib")
else()
    set(default_pipeline "${repository_root}/Bin/Logic/Module/${CMAKE_SHARED_LIBRARY_PREFIX}AssetsPipeline${CMAKE_SHARED_LIBRARY_SUFFIX}")
endif()
set(GAMESPROUT_TEST_PIPELINE_LIBRARY "${default_pipeline}" CACHE FILEPATH "Independently built production AssetsPipeline library")
add_library(PrefabPipeline SHARED IMPORTED)
set_target_properties(PrefabPipeline PROPERTIES IMPORTED_LOCATION "${GAMESPROUT_TEST_PIPELINE_LIBRARY}")
if(WIN32)
    get_filename_component(pipeline_library_dir "${GAMESPROUT_TEST_PIPELINE_LIBRARY}" DIRECTORY)
    set_target_properties(PrefabPipeline PROPERTIES IMPORTED_IMPLIB "${pipeline_library_dir}/AssetsPipeline.lib")
endif()
# Consume the real module, never compile a second copy of its private sources.
add_executable(PrefabPipelineChecks PrefabPipelineChecks.cpp)
target_include_directories(PrefabPipelineChecks PRIVATE
    "${gamesprout_root}/Module/AssetsPipeline" "${gamesprout_root}/Module/Interface"
    "${gamesprout_root}/GameSproutEditor/Api")
target_link_libraries(PrefabPipelineChecks PRIVATE PrefabPipeline Foundation::FoundationApi
    ObjectDocument::ObjectDocument Qt6::Core Qt6::Gui)
enable_testing()
add_test(NAME Assets.PrefabPreparedData COMMAND PrefabPipelineChecks)
set_tests_properties(Assets.PrefabPreparedData PROPERTIES TIMEOUT 30)
